// Gin 基于 httprouter，性能极高
package main

## 🏗️ **架构设计对比**

### **Gin 架构**

```go
// Gin 采用中间件链模式
package main

import (
    "github.com/gin-gonic/gin"
)

func main() {
    r := gin.New()

    // 中间件按顺序执行
    r.Use(gin.Logger())
    r.Use(gin.Recovery())
    r.Use(CorsMiddleware())
    r.Use(AuthMiddleware())

    // 路由分组
    api := r.Group("/api")
    {
        v1 := api.Group("/v1")
        {
            v1.GET("/users", GetUsers)
            v1.POST("/users", CreateUser)
        }
    }

    r.Run()
}

// 中间件定义
func AuthMiddleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        token := c.GetHeader("Authorization")
        if token == "" {
            c.AbortWithStatusJSON(401, gin.H{"error": "unauthorized"})
            return
        }
        c.Next() // 继续执行
    }
}
```

### **ASP.NET Core 架构**

```csharp
// ASP.NET Core 采用依赖注入和中间件管道
var builder = WebApplication.CreateBuilder(args);

// 依赖注入容器
builder.Services.AddScoped<IUserService, UserService>();
builder.Services.AddDbContext<AppDbContext>();
builder.Services.AddAuthentication(JwtBearerDefaults.AuthenticationScheme);

var app = builder.Build();

// 中间件管道（顺序很重要）
app.UseHttpsRedirection();
app.UseRouting();
app.UseAuthentication();  // 认证
app.UseAuthorization();   // 授权

app.MapControllers();     // 终结点路由

app.Run();

// 控制器方式
[ApiController]
[Route("api/[controller]")]
[Authorize]
public class UsersController : ControllerBase
{
    private readonly IUserService _userService;

    public UsersController(IUserService userService)
    {
        _userService = userService;
    }

    [HttpGet]
    public async Task<ActionResult<List<User>>> GetUsers()
    {
        var users = await _userService.GetAllAsync();
        return Ok(users);
    }

    [HttpPost]
    public async Task<ActionResult<User>> CreateUser(CreateUserDto dto)
    {
        var user = await _userService.CreateAsync(dto);
        return CreatedAtAction(nameof(GetUser), new { id = user.Id }, user);
    }
}
```

## 🎨 **适用场景对比**

### **Gin 最适合的场景**

```go
// 1. 高并发 API 网关
// 2. 微服务基础服务
// 3. 云原生/容器化应用
// 4. 边缘计算/IoT 服务
// 5. 需要快速启动和低内存的服务

// 示例：高并发计数器服务
type CounterService struct {
    mu    sync.RWMutex
    counts map[string]int64
}

func (s *CounterService) Increment(c *gin.Context) {
    key := c.Param("key")
    s.mu.Lock()
    s.counts[key]++
    val := s.counts[key]
    s.mu.Unlock()

    c.JSON(200, gin.H{"key": key, "count": val})
}
// 内存占用：~10MB，QPS：> 5万
```

### **ASP.NET Core 最适合的场景**

```csharp
// 1. 企业级复杂业务系统
// 2. 大型分布式应用
// 3. 需要丰富集成功能的项目
// 4. Windows 生态项目
// 5. 需要强类型和复杂业务逻辑

// 示例：企业 ERP 系统模块
[ApiController]
[Route("api/[controller]")]
public class OrderController : ControllerBase
{
    private readonly IMediator _mediator;
    private readonly IMapper _mapper;

    [HttpPost]
    [Authorize(Roles = "Admin,Manager")]
    public async Task<ActionResult<OrderDto>> CreateOrder(
        CreateOrderCommand command)
    {
        // 使用 MediatR 实现 CQRS
        var result = await _mediator.Send(command);

        // 使用 AutoMapper 进行对象映射
        var dto = _mapper.Map<OrderDto>(result);

        // 集成事件
        await _mediator.Publish(new OrderCreatedEvent(result.Id));

        return CreatedAtAction(nameof(GetOrder), new { id = result.Id }, dto);
    }
}
```

## 🎯 **选型建议**

### **选择 Gin 当：**

```go
// 1. 构建高并发 API 服务
// 2. 微服务基础架构
// 3. 资源受限环境（容器、边缘计算）
// 4. 团队熟悉 Go 语言
// 5. 需要快速迭代和简单部署

// 典型场景：BFF（Backend For Frontend）
r.GET("/api/bff/user/:id", func(c *gin.Context) {
    // 聚合多个微服务数据
    user := getUserFromService(c.Param("id"))
    orders := getOrdersFromService(c.Param("id"))
    profile := getProfileFromService(c.Param("id"))

    c.JSON(200, gin.H{
        "user": user,
        "orders": orders,
        "profile": profile,
    })
})
```

### **选择 ASP.NET Core 当：**

```csharp
// 1. 构建复杂业务系统（ERP、CRM、金融）
// 2. 已有 .NET 技术栈和团队
// 3. 需要丰富的企业级功能
// 4. Windows Azure 生态
// 5. 需要强类型和严格规范

// 典型场景：金融交易系统
[ApiController]
[Route("api/[controller]")]
[Authorize(Policy = "Trading")]
public class TradingController : ControllerBase
{
    [HttpPost("execute")]
    public async Task<ActionResult<TradeResult>> ExecuteTrade(
        [FromBody] TradeCommand command)
    {
        // 复杂业务验证
        var validator = new TradeValidator();
        var result = await validator.ValidateAsync(command);

        // 事务处理
        using var transaction = await _dbContext.Database
            .BeginTransactionAsync();

        try {
            // 多个数据库操作
            await _orderService.CreateOrder(command);
            await _accountService.UpdateBalance(command);
            await _auditService.LogTrade(command);

            await transaction.CommitAsync();
            return Ok(new TradeResult { Success = true });
        }
        catch
        {
            await transaction.RollbackAsync();
            throw;
        }
    }
}
```

### **最终建议：**

1. **如果是新项目、微服务、高并发场景**：选择 **Gin**
2. **如果是企业级复杂系统、已有 .NET 团队**：选择 **ASP.NET Core**
3. **如果项目需要快速开发，但资源有限**：选择 **Gin**
4. **如果需要丰富的企业功能和集成**：选择 **ASP.NET Core**

两个框架都很优秀，选择取决于具体场景、团队技术栈和项目需求！
