## 51. 什么是自定义模型绑定器？如何实现一个？

**自定义模型绑定器** 用于当默认模型绑定无法满足需求时，手动控制如何将 HTTP 请求数据转换为 .NET 类型。

**适用场景**：

- 解析自定义格式（如逗号分隔的 ID 列表 `"1,2,3"` → `List<int>`）
- 从请求头、Cookie 或其他位置绑定数据
- 处理加密/编码的参数

**实现步骤**：

```csharp
// 1. 实现 IModelBinder 接口
public class CommaSeparatedListBinder : IModelBinder
{
    public Task BindModelAsync(ModelBindingContext bindingContext)
    {
        var modelName = bindingContext.ModelName;
        var valueProviderResult = bindingContext.ValueProvider.GetValue(modelName);

        if (valueProviderResult == ValueProviderResult.None)
            return Task.CompletedTask;

        var rawValue = valueProviderResult.FirstValue;

        if (string.IsNullOrEmpty(rawValue))
            return Task.CompletedTask;

        // 将 "1,2,3" 转换为 List<int>
        var result = rawValue.Split(',')
            .Select(int.Parse)
            .ToList();

        bindingContext.Result = ModelBindingResult.Success(result);
        return Task.CompletedTask;
    }
}

// 2. 使用特性应用绑定器
public class SearchRequest
{
    [ModelBinder(BinderType = typeof(CommaSeparatedListBinder))]
    public List<int> ProductIds { get; set; }
}

[HttpGet("products")]
public IActionResult GetProducts([FromQuery] SearchRequest request)
{
    // GET /products?productIds=1,2,3 → request.ProductIds = [1, 2, 3]
    return Ok();
}

// 3. 更完整的实现（带错误处理）
public class DateTimeBinder : IModelBinder
{
    private readonly string _format;

    public DateTimeBinder(string format = "yyyy-MM-dd")
    {
        _format = format;
    }

    public Task BindModelAsync(ModelBindingContext bindingContext)
    {
        var value = bindingContext.ValueProvider.GetValue(bindingContext.ModelName);

        if (value == ValueProviderResult.None)
            return Task.CompletedTask;

        var strValue = value.FirstValue;

        if (DateTime.TryParseExact(strValue, _format, CultureInfo.InvariantCulture,
            DateTimeStyles.None, out var date))
        {
            bindingContext.Result = ModelBindingResult.Success(date);
        }
        else
        {
            bindingContext.ModelState.TryAddModelError(
                bindingContext.ModelName,
                $"日期格式必须是 {_format}");
        }

        return Task.CompletedTask;
    }
}

// 使用
public class ReportRequest
{
    [ModelBinder(typeof(DateTimeBinder), nameof(DateTimeBinder), "yyyy-MM-dd")]
    public DateTime StartDate { get; set; }
}
```

## 52. 如何启用或禁用 Controller 或 Action？使用什么特性？

## 53. 什么是 Controller 中的方法注入？与构造函数注入有何不同？

| 特性           | 构造函数注入             | 方法注入               |
| -------------- | ------------------------ | ---------------------- |
| **注入位置**   | 构造函数参数             | Action 方法参数        |
| **生命周期**   | 整个 Controller 生命周期 | 仅限该方法执行期间     |
| **使用场景**   | 大多数依赖               | 可选依赖、请求特定依赖 |
| **可空性**     | 通常是必需的             | 可以是不必需的         |
| **测试复杂度** | 较低                     | 较高（需构建参数）     |

**构造函数注入（标准方式）**：

```csharp
[ApiController]
[Route("api/[controller]")]
public class ProductsController : ControllerBase
{
    private readonly IProductService _productService;
    private readonly ILogger<ProductsController> _logger;

    // 依赖在 Controller 实例化时注入
    public ProductsController(IProductService productService, ILogger<ProductsController> logger)
    {
        _productService = productService;
        _logger = logger;
    }

    [HttpGet]
    public async Task<IActionResult> Get()
    {
        _logger.LogInformation("Getting products");
        return Ok(await _productService.GetAllAsync());
    }
}
```

**方法注入（通过 `[FromServices]` 特性）**：

```csharp
[ApiController]
[Route("api/[controller]")]
public class OrdersController : ControllerBase
{
    // 构造函数只注入必需的服务
    public OrdersController(IOrderService orderService)
    {
        _orderService = orderService;
    }

    [HttpGet]
    public async Task<IActionResult> Get(
        [FromServices] ICacheService cache,      // 仅此方法需要
        [FromServices] ILogger<OrdersController> logger)
    {
        logger.LogInformation("Using method injection");
        var cached = await cache.GetAsync("orders");
        return Ok(cached);
    }

    [HttpPost]
    public async Task<IActionResult> Create(Order order)
    {
        // 此方法不需要 cache 服务
        return Ok(await _orderService.CreateAsync(order));
    }
}
```

**何时使用方法注入**：

- 依赖仅在少数 Action 中使用
- 可选依赖（可空）
- 避免构造函数过于臃肿
- 不同 Action 需要不同的服务组合

## 54. 如何在 Controller Action 中实现验证？如何处理无效模型？

```csharp
    [Url]
    public string ProductUrl { get; set; }
}

// Controller 中的处理
[ApiController]
public class ProductsController : ControllerBase
{

    public IActionResult Create([FromBody] CreateProductRequest request)
```

```csharp
public class EventRequest
```

**使用 IValidatableObject 进行跨属性验证**：

```csharp
public class DateRangeRequest : IValidatableObject
{
    public DateTime StartDate { get; set; }
    public DateTime EndDate { get; set; }

    public IEnumerable<ValidationResult> Validate(ValidationContext validationContext)
    {
        if (EndDate <= StartDate)
        {
            yield return new ValidationResult(
                "结束日期必须晚于开始日期",
                new[] { nameof(EndDate) }
            );
        }

        if ((EndDate - StartDate).TotalDays > 365)
        {
            yield return new ValidationResult("日期范围不能超过365天");
        }
    }
}
```

**全局处理无效模型（自动返回 400）**：

```csharp
builder.Services.AddControllers()
    .ConfigureApiBehaviorOptions(options =>
    {

            return new BadRequestObjectResult(new

                Status = 400,
                Errors = errors
            });
        };
    });
```

## 55. 解释 ASP.NET Core 中 Controller 的生命周期

**Controller 默认是瞬态的（Transient）**：每个请求创建一个新的 Controller 实例。

**生命周期流程**：

```
1. 请求到达 → 路由匹配
2. ControllerFactory 创建 Controller 实例
   ├── 通过依赖注入解析构造函数参数
   └── 调用构造函数
3. 执行 Action 方法
   ├── 模型绑定
   ├── 验证（ModelState）
   ├── 过滤器管道执行
   └── Action 代码执行
4. Controller 实例被 Dispose（如果实现了 IDisposable）
5. 实例等待 GC 回收
```

**验证生命周期**：

```csharp
public class SampleController : ControllerBase, IDisposable
{
    private readonly ILogger<SampleController> _logger;

    public SampleController(ILogger<SampleController> logger)
    {
        _logger = logger;
        _logger.LogInformation("Controller 创建: {InstanceId}", GetHashCode());
    }

    [HttpGet]
    public IActionResult Get()
    {
        _logger.LogInformation("Action 执行中...");
        return Ok(new { InstanceId = GetHashCode() });
    }

    public void Dispose()
    {
        _logger.LogInformation("Controller 释放: {InstanceId}", GetHashCode());
    }
}
// 每次请求会看到不同的 InstanceId
```

**自定义 Controller 生命周期**：

```csharp
// 默认：瞬态（每个请求新实例）
builder.Services.AddTransient<ProductsController>();

// 改为 Scoped（同一请求内共享）
builder.Services.AddScoped<ProductsController>();

// 改为 Singleton（⚠️ 不推荐，会导致状态问题）
builder.Services.AddSingleton<ProductsController>();
```

**重要注意事项**：

- Controller 应该保持**无状态**（不存储请求相关数据在字段中）
- 不要将 Controller 注册为 Singleton（会导致内存泄漏和线程安全问题）
- 通过构造函数注入的服务生命周期由容器管理，与 Controller 无关

## 56. ControllerBase 和 Controller 的区别？何时使用哪个？

| 特性               | ControllerBase           | Controller                           |
| ------------------ | ------------------------ | ------------------------------------ |
| **继承自**         | 直接继承 Object          | 继承 ControllerBase                  |
| **View 支持**      | ❌ 不支持                | ✅ 支持 (View(), PartialView())      |
| **Razor 视图**     | ❌ 不支持                | ✅ 支持                              |
| **Web API 使用**   | ✅ 推荐                  | 可以但不推荐                         |
| **MVC Web 应用**   | 不适用                   | ✅ 标准使用                          |
| **添加的命名空间** | Microsoft.AspNetCore.Mvc | Microsoft.AspNetCore.Mvc + View 相关 |

**ControllerBase（用于 Web API）**：

```csharp
[ApiController]
[Route("api/[controller]")]
public class ProductsController : ControllerBase
{
    // 只有 API 相关方法
    [HttpGet]
    public IActionResult Get() => Ok();

    [HttpPost]
    public IActionResult Create() => CreatedAtAction(...);

    // ❌ 没有 View()、PartialView() 等方法
}
```

**Controller（用于 MVC 视图应用）**：

```csharp
public class HomeController : Controller
{
    // 拥有 ControllerBase 的所有 API 方法 + View 方法
    public IActionResult Index()
    {
        return View();  // 返回 Razor 视图
    }

    public IActionResult About()
    {
        return View(model);
    }

    public IActionResult Partial()
    {
        return PartialView("_MyPartial");
    }

    // 也可以返回 JSON
    [HttpGet("api/data")]
    public IActionResult ApiData()
    {
        return Json(new { name = "test" });
    }
}
```

**选择建议**：

| 应用类型     | 推荐基类         | 原因                                     |
| ------------ | ---------------- | ---------------------------------------- |
| REST API     | `ControllerBase` | 更轻量，不含 View 相关方法               |
| MVC Web 应用 | `Controller`     | 需要返回视图                             |
| 混合应用     | 分开使用         | API 用 ControllerBase，页面用 Controller |

## 57. 如何为所有 Controller 全局处理异常？

**方法一：使用异常处理中间件（推荐）**

```csharp
// 自定义异常处理中间件
public class GlobalExceptionMiddleware
{
    private readonly RequestDelegate _next;
    private readonly ILogger<GlobalExceptionMiddleware> _logger;
    private readonly IWebHostEnvironment _env;

    public GlobalExceptionMiddleware(RequestDelegate next, ILogger<GlobalExceptionMiddleware> logger, IWebHostEnvironment env)
    {
        _next = next;
        _logger = logger;
        _env = env;
    }

    public async Task InvokeAsync(HttpContext context)
    {
        try
        {
            await _next(context);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "未处理的异常: {Message}", ex.Message);
            await HandleExceptionAsync(context, ex);
        }
    }

    private Task HandleExceptionAsync(HttpContext context, Exception ex)
    {
        context.Response.ContentType = "application/json";
        context.Response.StatusCode = StatusCodes.Status500InternalServerError;

        var response = new
        {
            Status = 500,
            Title = "服务器内部错误",
            Detail = _env.IsDevelopment() ? ex.Message : "请稍后重试",
            TraceId = context.TraceIdentifier
        };

        return context.Response.WriteAsync(JsonSerializer.Serialize(response));
    }
}

// 注册中间件（放在最前面）
app.UseMiddleware<GlobalExceptionMiddleware>();
```

**方法二：使用 ExceptionFilter（更细粒度控制）**

```csharp
public class GlobalExceptionFilter : IExceptionFilter
{
    private readonly ILogger<GlobalExceptionFilter> _logger;
    private readonly IWebHostEnvironment _env;

    public GlobalExceptionFilter(ILogger<GlobalExceptionFilter> logger, IWebHostEnvironment env)
    {
        _logger = logger;
        _env = env;
    }

    public void OnException(ExceptionContext context)
    {
        _logger.LogError(context.Exception, "Action 异常");

        var response = new
        {
            Status = 500,
            Message = _env.IsDevelopment() ? context.Exception.Message : "服务器错误"
        };

        context.Result = new ObjectResult(response)
        {
            StatusCode = StatusCodes.Status500InternalServerError
        };

        context.ExceptionHandled = true;
    }
}

// 全局注册
builder.Services.AddControllers(options =>
{
    options.Filters.Add<GlobalExceptionFilter>();
});
```

```csharp
{

    {

        {
```

```csharp
        }
        catch (NotFoundException ex)
        {

        }
    }
```

## 58. 什么是结构化日志？为什么要使用它？

**结构化日志** 将日志作为**键值对**记录，而不是纯文本字符串，便于查询、分析和报警。

**对比示例**：

```csharp
// 非结构化日志（纯文本）
_logger.LogInformation($"用户 {user.Id} 在 {DateTime.Now} 登录成功");
// 输出: "用户 123 在 2024-01-15 10:30:00 登录成功"

// 结构化日志（键值对）
_logger.LogInformation("用户 {UserId} 在 {LoginTime} 登录成功", user.Id, DateTime.Now);
// 输出 + 元数据: { "UserId": 123, "LoginTime": "2024-01-15T10:30:00", "Message": "用户 123 登录成功" }
```

**为什么要使用结构化日志**：

| 优势             | 说明                                             |
| ---------------- | ------------------------------------------------ |
| **易于查询**     | 在 Seq、Elasticsearch 中可用 `UserId = 123` 查询 |
| **支持聚合分析** | 可统计 `SELECT COUNT(*) GROUP BY EventType`      |
| **关联追踪**     | 通过 `CorrelationId` 追踪完整请求链路            |
| **自动报警**     | 监控 `ErrorCount > 10` 触发报警                  |
| **性能更好**     | 延迟计算，仅在需要时格式化字符串                 |
| **GDPR 合规**    | 自动过滤或掩码敏感字段                           |

**结构化日志实践（使用 Serilog）**：

```csharp
// 配置 Serilog
Log.Logger = new LoggerConfiguration()
    .Enrich.FromLogContext()
    .Enrich.WithProperty("Application", "MyApp")
    .Enrich.WithMachineName()
    .WriteTo.Console(new JsonFormatter())
    .WriteTo.Seq("http://localhost:5341")
    .CreateLogger();

builder.Host.UseSerilog();

// 记录结构化日志
public class OrderService
{
    private readonly ILogger<OrderService> _logger;

    public async Task<Order> CreateOrderAsync(OrderRequest request)
    {
        using var scope = _logger.BeginScope(new Dictionary<string, object>
        {

        });

        _logger.LogInformation("开始创建订单，产品: {ProductId}, 数量: {Quantity}",

        {

            _logger.LogInformation("订单创建成功，订单号: {OrderId}, 金额: {Amount:C}",
                order.Id, order.TotalAmount);

            return order;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "订单创建失败，产品: {ProductId}, 用户: {UserId}",
```

## 59. 如何记录传入 HTTP 请求的请求和响应数据？

**方法一：自定义中间件（推荐）**

```csharp
public class RequestResponseLoggingMiddleware
{
    private readonly RequestDelegate _next;
    private readonly ILogger<RequestResponseLoggingMiddleware> _logger;

    public RequestResponseLoggingMiddleware(RequestDelegate next, ILogger<RequestResponseLoggingMiddleware> logger)
    {
        _next = next;
        _logger = logger;
    }

    public async Task InvokeAsync(HttpContext context)
    {
        // 记录请求
        await LogRequest(context);

        // 拦截响应
        var originalBodyStream = context.Response.Body;
        using var responseBody = new MemoryStream();
        context.Response.Body = responseBody;

        // 执行后续中间件
        await _next(context);

        // 记录响应
        await LogResponse(context, responseBody);

        // 恢复响应流
        await responseBody.CopyToAsync(originalBodyStream);
    }

    private async Task LogRequest(HttpContext context)
    {
        context.Request.EnableBuffering();

        var body = await new StreamReader(context.Request.Body).ReadToEndAsync();
        context.Request.Body.Position = 0;

        _logger.LogInformation(@"
HTTP 请求:
Method: {Method}
Path: {Path}
QueryString: {QueryString}
Headers: {Headers}
Body: {Body}",
            context.Request.Method,
            context.Request.Path,
            context.Request.QueryString,
            context.Request.Headers.ToDictionary(h => h.Key, h => h.Value.ToString()),
            body);
    }

    private async Task LogResponse(HttpContext context, MemoryStream responseBody)
    {
        responseBody.Position = 0;
        var body = await new StreamReader(responseBody).ReadToEndAsync();
        responseBody.Position = 0;

        _logger.LogInformation(@"
HTTP 响应:
Status: {StatusCode}
Headers: {Headers}
Body: {Body}",
            context.Response.StatusCode,
            context.Response.Headers.ToDictionary(h => h.Key, h => h.Value.ToString()),
            body);
    }
}

app.UseMiddleware<RequestResponseLoggingMiddleware>();
```

## 60. 如何处理日志中的敏感数据以确保安全性和合规性（如 GDPR）？

**方法一：使用 LogContext 自动脱敏**

```csharp
public class SensitiveDataMasker
{
    private static readonly HashSet<string> SensitiveKeys = new()
    {
        "password", "token", "secret", "creditcard", "ssn", "email"
    };

    public static string MaskSensitiveData(string key, string value)
    {
        if (SensitiveKeys.Any(k => key.Contains(k, StringComparison.OrdinalIgnoreCase)))
        {
            return "***REDACTED***";
        }

        // 对邮箱部分脱敏
        if (key.Equals("email", StringComparison.OrdinalIgnoreCase) && value.Contains('@'))
        {
            var parts = value.Split('@');
            var masked = parts[0].Length > 2
                ? parts[0].Substring(0, 2) + "***"
                : "***";
            return $"{masked}@{parts[1]}";
        }

        return value;
    }
}

// 自定义日志 Enricher
public class SensitiveDataEnricher : ILogEventEnricher
{
    public void Enrich(LogEvent logEvent, ILogEventPropertyFactory propertyFactory)
    {
        foreach (var property in logEvent.Properties.ToList())
        {
            var masked = SensitiveDataMasker.MaskSensitiveData(property.Key, property.Value.ToString());
            logEvent.AddOrUpdateProperty(propertyFactory.CreateProperty(property.Key, masked));
        }
    }
}

// 配置 Serilog
Log.Logger = new LoggerConfiguration()
    .Enrich.With<SensitiveDataEnricher>()
    .WriteTo.File("logs/app-.txt", rollingInterval: RollingInterval.Day)
    .CreateLogger();
```

**方法四：实现自定义 ILoggerProvider 进行脱敏**

```csharp
public class RedactingLoggerProvider : ILoggerProvider
{
    private readonly ILoggerProvider _innerProvider;

    public RedactingLoggerProvider(ILoggerProvider innerProvider)
    {
        _innerProvider = innerProvider;
    }

    public ILogger CreateLogger(string categoryName)
    {
        var innerLogger = _innerProvider.CreateLogger(categoryName);
        return new RedactingLogger(innerLogger);
    }

    public void Dispose() => _innerProvider.Dispose();
}

public class RedactingLogger : ILogger
{
    private readonly ILogger _inner;

    public RedactingLogger(ILogger inner) => _inner = inner;

    public void Log<TState>(LogLevel logLevel, EventId eventId, TState state, Exception exception, Func<TState, Exception, string> formatter)
    {
        // 重写 formatter 来脱敏
        string newFormatter(TState s, Exception e) =>
            RedactSensitiveData(formatter(s, e));

        _inner.Log(logLevel, eventId, state, exception, (s, e) => newFormatter((TState)s, e));
    }

    private string RedactSensitiveData(string message)
    {
        // 正则匹配并替换敏感信息
        message = Regex.Replace(message, @"password""?\s*:\s*""[^""]+""", "password\": \"***\"", RegexOptions.IgnoreCase);
        message = Regex.Replace(message, @"\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b", "[EMAIL_REDACTED]");
        return message;
    }

    public bool IsEnabled(LogLevel logLevel) => _inner.IsEnabled(logLevel);
    public IDisposable BeginScope<TState>(TState state) => _inner.BeginScope(state);
}
```
