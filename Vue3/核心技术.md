# 🚀 **Vue 3 核心技术全景**

除了 Props 和 Slot，Vue 3 还有以下核心技术：

## 📋 **核心 API 分类**

### **一、组合式 API (Composition API)**

```typescript
// 响应式核心
import { ref, reactive, computed, watch, watchEffect } from "vue";

// 生命周期
import { onMounted, onUpdated, onUnmounted } from "vue";

// 依赖注入
import { provide, inject } from "vue";
```

### **二、响应式系统**

#### **1. ref() - 响应式基本类型**

```typescript
const count = ref(0);
count.value++; // 触发更新

// 模板自动解包
// <div>{{ count }}</div>
```

#### **2. reactive() - 响应式对象**

```typescript
const user = reactive({
  name: "张三",
  age: 25,
  address: {
    city: "北京",
  },
});
user.age = 26; // 深层响应
```

#### **3. computed() - 计算属性**

```typescript
const fullName = computed(() => `${firstName.value} ${lastName.value}`);

// 可写计算属性
const modelValue = computed({
  get() {
    return props.modelValue;
  },
  set(value) {
    emit("update:modelValue", value);
  },
});
```

#### **4. watch() / watchEffect() - 侦听器**

```typescript
// watch - 精确侦听
watch(
  () => user.id, // 侦听源
  (newId, oldId) => {
    console.log(`ID 从 ${oldId} 变为 ${newId}`);
  },
  { immediate: true, deep: true },
);

// watchEffect - 自动追踪依赖
watchEffect(() => {
  console.log(`用户: ${user.name}, 年龄: ${user.age}`);
});
```

### **三、组件通信**

#### **1. defineEmits() - 子传父**

```vue
<script setup>
// 定义事件
const emit = defineEmits<{
  'update:modelValue': [value: string]
  'submit': [data: FormData]
  'cancel': []
}>()

// 触发事件
function handleSubmit() {
  emit('submit', formData)
}
</script>
```

#### **2. provide() / inject() - 依赖注入**

```typescript
// 祖先组件提供
import { provide } from "vue";

const user = ref({ name: "张三" });
provide("user", user); // 提供响应式数据

// 后代组件注入
import { inject } from "vue";
const user = inject("user", defaultValue);
```

### **四、状态管理**

#### **Pinia (Vuex 的替代品)**

```typescript
// store/user.ts
import { defineStore } from "pinia";

export const useUserStore = defineStore("user", {
  state: () => ({
    name: "",
    token: "",
    permissions: [],
  }),
  getters: {
    isLoggedIn: (state) => !!state.token,
  },
  actions: {
    async login(credentials) {
      const { data } = await api.login(credentials);
      this.token = data.token;
    },
  },
});

// 组件中使用
import { useUserStore } from "@/stores/user";
const userStore = useUserStore();
userStore.login({ username, password });
```

### **五、路由管理**

#### **Vue Router 4**

```typescript
import { createRouter, createWebHistory } from "vue-router";

const router = createRouter({
  history: createWebHistory(),
  routes: [
    {
      path: "/user/:id",
      component: UserDetail,
      props: true, // 将路由参数作为 props 传递
      meta: { requiresAuth: true },
    },
  ],
});

// 组件中使用
import { useRoute, useRouter } from "vue-router";
const route = useRoute(); // 当前路由信息
const router = useRouter(); // 路由实例

router.push("/user/123"); // 导航
```

### **六、模板特性**

#### **1. 指令系统**

```vue
<template>
  <!-- 内置指令 -->
  <div v-if="show" v-html="content"></div>
  <ul>
    <li v-for="item in list" :key="item.id">{{ item.name }}</li>
  </ul>

  <!-- 自定义指令 -->
  <div v-focus></div>
  <div v-click-outside="closeMenu"></div>
</template>

<script setup>
// 自定义指令
const vFocus = {
  mounted: (el) => el.focus(),
};
</script>
```

#### **2. 组件引用**

```vue
<template>
  <!-- 模板引用 -->
  <input ref="inputRef" />
  <ChildComponent ref="childRef" />
</template>

<script setup>
import { ref, onMounted } from 'vue'

// 获取 DOM 引用
const inputRef = ref<HTMLInputElement>()
const childRef = ref<InstanceType<typeof ChildComponent>>()

onMounted(() => {
  inputRef.value?.focus()  // 访问 DOM
  childRef.value?.someMethod()  // 访问子组件方法
})
</script>
```

### **七、高级特性**

#### **1. Teleport - 传送门**

```vue
<template>
  <button @click="showModal = true">打开模态框</button>

  <!-- 将内容渲染到 body 下 -->
  <Teleport to="body">
    <Modal v-if="showModal" @close="showModal = false">
      <h2>标题</h2>
      <p>内容</p>
    </Modal>
  </Teleport>
</template>
```

#### **2. Suspense - 异步组件**

```vue
<template>
  <Suspense>
    <!-- 显示异步组件 -->
    <template #default>
      <AsyncComponent />
    </template>

    <!-- 加载状态 -->
    <template #fallback>
      <div>加载中...</div>
    </template>
  </Suspense>
</template>

<script setup>
// 异步组件
const AsyncComponent = defineAsyncComponent(
  () => import("./AsyncComponent.vue"),
);
</script>
```

#### **3. 自定义渲染器**

```typescript
// 创建自定义渲染器（用于 Canvas、WebGL 等）
import { createRenderer } from "vue";

const { createApp } = createRenderer({
  // 自定义渲染逻辑
});
```

### **八、性能优化**

#### **1. 响应式工具函数**

```typescript
import {
  toRef,
  toRefs,
  isRef,
  unref,
  shallowRef,
  shallowReactive,
  markRaw,
} from "vue";

// toRefs - 解构保持响应性
const { name, age } = toRefs(user);

// shallowReactive - 浅层响应式
const shallowObj = shallowReactive({
  nested: { data: "不会响应" }, // 嵌套对象不是响应式的
});

// markRaw - 标记为非响应式
const staticData = markRaw({ id: 1 }); // 不会被代理
```

#### **2. 组件优化**

```vue
<template>
  <!-- KeepAlive 缓存组件 -->
  <KeepAlive :include="['ComponentA']" :max="10">
    <component :is="currentComponent" />
  </KeepAlive>
</template>

<script setup>
// defineOptions - 定义组件选项
defineOptions({
  name: "MyComponent",
  inheritAttrs: false,
});

// defineExpose - 暴露组件方法
const someMethod = () => {};
defineExpose({ someMethod });
</script>
```

## 🚀 **实战重点**

```typescript
// 1. 自定义组合式函数 (Composables)
export function useMouse() {
  const x = ref(0);
  const y = ref(0);

  function update(e: MouseEvent) {
    x.value = e.pageX;
    y.value = e.pageY;
  }

  onMounted(() => window.addEventListener("mousemove", update));
  onUnmounted(() => window.removeEventListener("mousemove", update));

  return { x, y };
}
```
